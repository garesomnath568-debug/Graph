
#include <iostream>
#include <algorithm>
using namespace std;

struct Edge
{
    int src, dest, weight;
};

int parent[100];

// Find function
int find(int i)
{
    if (parent[i] == i)
        return i;

    return parent[i] = find(parent[i]);
}

// Union function
void Union(int u, int v)
{
    int a = find(u);
    int b = find(v);

    parent[a] = b;
}

// Kruskal's Algorithm
void Kruskal(Edge edges[], int V, int E)
{
    sort(edges, edges + E,
         [](Edge a, Edge b)
         {
             return a.weight < b.weight;
         });

    for (int i = 0; i < V; i++)
        parent[i] = i;

    int totalCost = 0;
    int count = 0;

    cout << "\nSelected Pipelines:\n";

    for (int i = 0; i < E && count < V - 1; i++)
    {
        int u = edges[i].src;
        int v = edges[i].dest;

        if (find(u) != find(v))
        {
            cout << u << " -- " << v
                 << " : Rs. " << edges[i].weight << endl;

            Union(u, v);

            totalCost += edges[i].weight;
            count++;
        }
    }

    if (count != V - 1)
    {
        cout << "\nAll tanks cannot be connected!";
        return;
    }

    cout << "\nMinimum Installation Cost = Rs. "
         << totalCost << endl;
}

int main()
{
    int V, E;
    Edge edges[100];

    cout << "Enter number of water tanks: ";
    cin >> V;

    cout << "Enter number of pipelines: ";
    cin >> E;

    cout << "\nEnter pipeline details "
         << "(Source Destination Cost):\n";

    for (int i = 0; i < E; i++)
    {
        cin >> edges[i].src
            >> edges[i].dest
            >> edges[i].weight;
    }

    Kruskal(edges, V, E);

    return 0;
}
